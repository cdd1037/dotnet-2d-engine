cmake_minimum_required(VERSION 3.20)
project(GameAuthoringLab VERSION 0.1.0 LANGUAGES C CXX)
option(GAL_HEADLESS_ONLY "Build contract tests without SDL; cannot render or play audio" OFF)
add_library(gal SHARED native/src/gal.cpp)
target_include_directories(gal PUBLIC native/include)
target_compile_definitions(gal PRIVATE GAL_BUILD)
target_compile_features(gal PRIVATE cxx_std_17)
set_target_properties(gal PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(UNIX AND NOT APPLE)
 target_link_options(gal PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/native/gal.exports")
endif()
if(MSVC)
 target_compile_options(gal PRIVATE /W4 /WX)
else()
 target_compile_options(gal PRIVATE -Wall -Wextra -Werror)
endif()
if(GAL_HEADLESS_ONLY)
 target_sources(gal PRIVATE native/src/backend_headless.cpp)
else()
 find_package(SDL3 3.2.28 CONFIG REQUIRED)
 target_sources(gal PRIVATE native/src/backend_sdl.cpp)
 target_link_libraries(gal PRIVATE SDL3::SDL3)
endif()
find_package(Threads REQUIRED)
target_link_libraries(gal PRIVATE Threads::Threads)
enable_testing()
add_executable(gal_native_tests tests/native_tests.cpp)
target_link_libraries(gal_native_tests PRIVATE gal Threads::Threads)
add_test(NAME native_contract COMMAND gal_native_tests)
add_executable(gal_backend_contract_tests tests/backend_contract_tests.cpp native/src/gal.cpp)
target_include_directories(gal_backend_contract_tests PRIVATE native/include native/src)
target_compile_definitions(gal_backend_contract_tests PRIVATE GAL_BUILD)
target_compile_features(gal_backend_contract_tests PRIVATE cxx_std_17)
target_link_libraries(gal_backend_contract_tests PRIVATE Threads::Threads)
add_test(NAME backend_contract COMMAND gal_backend_contract_tests)
add_executable(gal_c_consumer tests/c_consumer.c)
target_compile_features(gal_c_consumer PRIVATE c_std_11)
target_link_libraries(gal_c_consumer PRIVATE gal)
add_test(NAME c_abi_consumer COMMAND gal_c_consumer)

add_executable(gal_input_state_tests tests/input_state_tests.cpp)
target_include_directories(gal_input_state_tests PRIVATE native/include native/src)
target_compile_features(gal_input_state_tests PRIVATE cxx_std_17)
add_test(NAME input_state_contract COMMAND gal_input_state_tests)

option(GAL_ENABLE_RMLUI "Enable reversible RmlUi settings UI experiment" OFF)
set(GAL_RMLUI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/RmlUi-ba95ffe8bfb6370efb2cdcca927eaad4710c5413" CACHE PATH "Pinned RmlUi 6.3 source")
set(GAL_RMLUI_CORE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/rmlui-build/librmlui.a" CACHE FILEPATH "Built RmlUi core static library")
if(GAL_ENABLE_RMLUI)
 if(GAL_HEADLESS_ONLY)
  message(FATAL_ERROR "RmlUi experiment requires SDL GPU")
 endif()
 find_package(SDL3_image CONFIG REQUIRED)
 find_package(Freetype REQUIRED)
 target_sources(gal PRIVATE native/src/ui_rml.cpp "${GAL_RMLUI_SOURCE}/Backends/RmlUi_Platform_SDL.cpp" "${GAL_RMLUI_SOURCE}/Backends/RmlUi_Renderer_SDL_GPU.cpp")
 target_include_directories(gal PRIVATE "${GAL_RMLUI_SOURCE}/Include" "${GAL_RMLUI_SOURCE}/Backends")
 target_compile_definitions(gal PRIVATE GAL_ENABLE_RMLUI RMLUI_STATIC_LIB RMLUI_SDL_VERSION_MAJOR=3)
 target_link_libraries(gal PRIVATE "${GAL_RMLUI_CORE}" SDL3_image::SDL3_image Freetype::Freetype)
endif()
